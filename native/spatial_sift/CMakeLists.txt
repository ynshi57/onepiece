cmake_minimum_required(VERSION 3.16)
project(vqasee_spatial_sift LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_LIST core,imgproc,features2d,flann CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_PERF_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_opencv_apps OFF CACHE BOOL "" FORCE)
set(BUILD_JAVA OFF CACHE BOOL "" FORCE)
set(BUILD_opencv_python2 OFF CACHE BOOL "" FORCE)
set(BUILD_opencv_python3 OFF CACHE BOOL "" FORCE)
set(WITH_IPP OFF CACHE BOOL "" FORCE)
set(WITH_ITT OFF CACHE BOOL "" FORCE)
set(WITH_OPENCL OFF CACHE BOOL "" FORCE)
set(WITH_ADE OFF CACHE BOOL "" FORCE)
set(WITH_PROTOBUF OFF CACHE BOOL "" FORCE)
set(WITH_FFMPEG OFF CACHE BOOL "" FORCE)
set(WITH_GSTREAMER OFF CACHE BOOL "" FORCE)
set(WITH_AVFOUNDATION OFF CACHE BOOL "" FORCE)
set(WITH_CAP_IOS OFF CACHE BOOL "" FORCE)
set(WITH_QT OFF CACHE BOOL "" FORCE)
set(WITH_GTK OFF CACHE BOOL "" FORCE)
set(WITH_VTK OFF CACHE BOOL "" FORCE)
set(WITH_OPENEXR OFF CACHE BOOL "" FORCE)
set(OPENCV_SKIP_PYTHON_LOADER ON CACHE BOOL "" FORCE)
foreach(OPTION JPEG PNG TIFF WEBP OPENJPEG JASPER ZLIB LAPACK EIGEN)
  set(WITH_${OPTION} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(vendor/opencv opencv EXCLUDE_FROM_ALL)
add_library(spatial_sift SHARED bridge.cpp line_bridge.cpp)
target_include_directories(spatial_sift PRIVATE vendor/opencv/modules/core/include vendor/opencv/modules/imgproc/include vendor/opencv/modules/features2d/include vendor/opencv/modules/flann/include ${CMAKE_BINARY_DIR})
target_link_libraries(spatial_sift PRIVATE opencv_features2d opencv_imgproc)
